   483→function ProfileEditor({ user, editing, setEditing }: { user: any; editing: boolean; setEditing: (v: boolean) => void }) {
   484→  const [form, setForm] = useState({
   485→    name: user.name || "",
   486→    phone: user.phone || "",
   487→    country: user.country || "",
   488→    bio: user.profile?.bio || "",
   489→    occupation: user.profile?.occupation || "",
   490→    skills: user.profile?.skills ? JSON.parse(user.profile.skills).join(", ") : "",
   491→    languages: user.profile?.languages ? JSON.parse(user.profile.languages).join(", ") : "",
   492→  });
   493→  const [saving, setSaving] = useState(false);
   494→
   495→  const handleSave = async () => {
   496→    setSaving(true);
   497→    try {
   498→      const res = await fetch("/api/auth/me", {
   499→        method: "PATCH",
   500→        headers: { "Content-Type": "application/json" },
   501→        body: JSON.stringify({
   502→          name: form.name,
   503→          phone: form.phone,
   504→          country: form.country,
   505→          bio: form.bio,
   506→          occupation: form.occupation,
   507→          skills: form.skills.split(",").map((s) => s.trim()).filter(Boolean),
   508→          languages: form.languages.split(",").map((s) => s.trim()).filter(Boolean),
   509→        }),
   510→      });
   511→      if (res.ok) {
   512→        toast.success("Profil mis à jour !");
   513→        setEditing(false);
   514→      } else {
   515→        toast.error("Erreur lors de la mise à jour");
   516→      }
   517→    } catch {
   518→      toast.error("Erreur réseau");
   519→    }
   520→    setSaving(false);
   521→  };
   522→
   523→  return (
   524→    <div className="bg-white rounded-xl p-6 border border-[#E8ECF1] shadow-premium">
   525→      <div className="flex items-center justify-between mb-6">
   526→        <h3 className="font-display font-bold text-[#003366] text-sm uppercase tracking-wider">Mon profil</h3>
   527→        {!editing ? (
   528→          <button onClick={() => setEditing(true)} className="text-[12px] font-semibold text-[#003366] hover:underline flex items-center gap-1.5">
   529→            <Edit3 className="w-3.5 h-3.5" /> Modifier
   530→          </button>
   531→        ) : (
   532→          <div className="flex gap-2">
   533→            <button onClick={() => setEditing(false)} className="text-[12px] font-semibold text-[#5C6573] hover:underline flex items-center gap-1.5">
   534→              <X className="w-3.5 h-3.5" /> Annuler
   535→            </button>
   536→            <button onClick={handleSave} disabled={saving} className="text-[12px] font-semibold text-[#FFD700] hover:underline flex items-center gap-1.5">
   537→              <Save className="w-3.5 h-3.5" /> {saving ? "..." : "Enregistrer"}
   538→            </button>
   539→          </div>
   540→        )}
   541→      </div>
   542→
   543→      <div className="grid sm:grid-cols-2 gap-4">
   544→        <Field label="Nom complet" value={form.name} onChange={(v) => setForm({ ...form, name: v })} editing={editing} />
   545→        <Field label="Email" value={user.email} editing={false} />
   546→        <Field label="Téléphone" value={form.phone} onChange={(v) => setForm({ ...form, phone: v })} editing={editing} />
   547→        <Field label="Pays" value={form.country} onChange={(v) => setForm({ ...form, country: v })} editing={editing} />
   548→        <Field label="Occupation" value={form.occupation} onChange={(v) => setForm({ ...form, occupation: v })} editing={editing} />
   549→        <Field label="Compétences (séparées par virgules)" value={form.skills} onChange={(v) => setForm({ ...form, skills: v })} editing={editing} />
   550→        <Field label="Langues (séparées par virgules)" value={form.languages} onChange={(v) => setForm({ ...form, languages: v })} editing={editing} />
   551→        <div className="sm:col-span-2">
   552→          <label className="block text-[11px] text-[#5C6573] mb-1.5 font-medium uppercase tracking-wider">Bio</label>
   553→          {editing ? (
   554→            <textarea value={form.bio} onChange={(e) => setForm({ ...form, bio: e.target.value })} rows={3} className="input-shine rounded-md px-3 py-2 text-sm w-full resize-none" />
   555→          ) : (
   556→            <p className="text-[13px] text-[#003366] min-h-[2.5rem]">{form.bio || "—"}</p>
   557→          )}
   558→        </div>
   559→      </div>
   560→    </div>
   561→  );
   562→}

   564→function Field({ label, value, onChange, editing }: { label: string; value: string; onChange?: (v: string) => void; editing: boolean }) {

   566→    <div>

   569→        <input value={value} onChange={(e) => onChange(e.target.value)} className="input-shine rounded-md px-3 py-2 text-sm w-full" />
   570→      ) : (
   571→        <p className="text-[13px] text-[#003366] min-h-[2.5rem]">{value || "—"}</p>
