   206→   206→function UsersManager() {

   208→   208→  const [users, setUsers] = useState<any[]>([]);
   209→   209→  const [loading, setLoading] = useState(true);
   210→   210→
   211→   211→  const load = async () => {
   212→   212→    setLoading(true);
   213→   213→    const res = await fetch(`/api/admin/users?search=${encodeURIComponent(search)}`);
   214→   214→    const data = await res.json();
   215→   215→    if (!data.error) setUsers(data.users);
   216→   216→    setLoading(false);
   217→   217→  };

   219→   219→  useEffect(() => {
